class ServerThread(threading.Thread):
    """Thread pour faire tourner Flask sans bloquer la fenêtre"""
    def __init__(self, app, host, port):
        threading.Thread.__init__(self)
        self.server = make_server(host, port, app)
        self.ctx = app.app_context()
        self.ctx.push()

    def run(self):
        self.server.serve_forever()

    def shutdown(self):
        self.server.shutdown()

server_thread = None

def get_ip():
    s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    try: s.connect(('8.8.8.8', 80)); IP = s.getsockname()[0]
    except: IP = '127.0.0.1'
    finally: s.close()
    return IP

def start_server():
    global server_thread, AUTH_USER, AUTH_PASS, ROOT_DIR
    
    # Récupérer les infos des champs
    user = entry_user.get()
    pwd = entry_pass.get()
    
    if not user or not pwd:
        messagebox.showerror("Erreur", "Veuillez remplir le user et le mot de passe")
        return

    AUTH_USER = user
    AUTH_PASS = pwd
    ip = get_ip()
    port = 8000
    
    btn_start.config(state="disabled", text="Serveur en ligne 🚀")
    btn_stop.config(state="normal")
    
    # Affichage dans la console GUI
    log_area.insert(tk.END, f"\n[+] Serveur démarré !\n")
    log_area.insert(tk.END, f"[i] IP : http://{ip}:{port}\n")
    log_area.insert(tk.END, f"[i] Login : {user} / {pwd}\n")
    log_area.see(tk.END)
    
    # Lancement Flask dans un thread séparé
    server_thread = ServerThread(app, "0.0.0.0", port)
    server_thread.start()
    
    # Ouvrir le navigateur localement
    webbrowser.open(f"http://127.0.0.1:{port}")

def stop_server():
    global server_thread
    if server_thread:
        log_area.insert(tk.END, "\n[!] Arrêt du serveur...\n")
        server_thread.shutdown()
        server_thread = None
        
    btn_start.config(state="normal", text="Démarrer le Serveur")
    btn_stop.config(state="disabled")
    log_area.insert(tk.END, "[!] Serveur arrêté.\n")
